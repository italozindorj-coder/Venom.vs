local function resolveBg(url, idx)
	if type(url) ~= "string" or url == "" then return nil end
	-- Roblox asset ids work as Image directly
	if url:find("^rbxassetid://") then return url end
	if type(getcustomasset) ~= "function" then return nil end
	local folder = "k2_riddler_bg"
	pcall(function()
		local mf = makefolder or (syn and syn.makefolder)
		if mf then pcall(mf, folder) end
	end)
	local ext = (url:find("%.png") and ".png") or ".jpg"
	local fname = folder .. "/bg" .. tostring(idx) .. ext
	local has = false
	pcall(function() if isfile and isfile(fname) then has = true end end)
	if not has then
		local ok, data = pcall(function() return game:HttpGet(url) end)
		if not ok or type(data) ~= "string" or #data < 500 then
			-- fallback request APIs
			pcall(function()
				local req = (syn and syn.request) or (http and http.request) or http_request or request
				if not req then return end
				local res = req({ Url = url, Method = "GET" })
				if res and type(res.Body) == "string" and #res.Body > 500 then
					data = res.Body
					ok = true
				end
			end)
		end
		if ok and type(data) == "string" and #data > 500 then
			pcall(function()
				local wf = writefile or (syn and syn.writefile)
				if wf then wf(fname, data) end
			end)
		end
	end
	local ok2, asset = pcall(function() return getcustomasset(fname) end)
	if ok2 and type(asset) == "string" and asset ~= "" then return asset end
	return nil
end

local _cachedBox = nil
local function formatCode(c)
	c = tostring(c or ""):gsub("%s+", ""):upper()
	return c

local function isGuiVisible(obj)
	local cur = obj
	while cur and cur ~= game do
		if cur:IsA("GuiObject") then
			if cur.Visible == false then return false end
			if cur.AbsoluteSize.X <= 0 or cur.AbsoluteSize.Y <= 0 then return false end
		end
		cur = cur.Parent
	end
	return true
end

local function _isCodeBox(obj)
	if not obj or not obj:IsA("TextBox") then return false end
	if ScreenGui and obj:IsDescendantOf(ScreenGui) then return false end
	local okVis = true
	pcall(function() okVis = isGuiVisible(obj) end)
	if not okVis then return false end
	local hint = ((obj.PlaceholderText or "") .. " " .. obj.Name):lower()
	if hint:find("code", 1, true) or hint:find("redeem", 1, true)
		or hint:find("enter", 1, true) or hint:find("type", 1, true)
		or hint:find("here", 1, true) then
		return true
	end
	local sz = obj.AbsoluteSize
	return sz.X >= 80 and sz.Y >= 20 and sz.Y <= 80
end

local function findCodeTextBox()
	local pg = LP:FindFirstChild("PlayerGui")
	if pg then
		local codes = pg:FindFirstChild("Codes")
		if codes then
			local inner = codes:FindFirstChild("Codes") or codes
			local redeem = inner and inner:FindFirstChild("CodeRedeem")
			local tbRedeem = redeem and redeem:FindFirstChild("TextBox")
			if tbRedeem and tbRedeem:IsA("TextBox") then
				if not tbRedeem.Active then
					local t0 = os.clock()
					while os.clock() - t0 < 0.8 and tbRedeem.Parent and not tbRedeem.Active do
						task.wait(0.05)
					end
				end
				_cachedBox = tbRedeem
				return tbRedeem
			end
			local main = inner and inner:FindFirstChild("Main")
			local tb = main and main:FindFirstChild("TextBox")
			if tb and tb:IsA("TextBox") then
				_cachedBox = tb
				return tb
			end
			for _, d in ipairs(codes:GetDescendants()) do
				if d:IsA("TextBox") and d:FindFirstAncestor("CodeRedeem") then
					_cachedBox = d
					return d
				end
			end
			for _, d in ipairs(codes:GetDescendants()) do
				if d:IsA("TextBox") then
					_cachedBox = d
					return d
				end
			end
		end
	end
	if _cachedBox then
		local still = false
		pcall(function()
			still = _cachedBox.Parent ~= nil and _isCodeBox(_cachedBox)
		end)
		if still then return _cachedBox end
		_cachedBox = nil
	end
	local focused = nil
	pcall(function() focused = UIS:GetFocusedTextBox() end)
	if focused and _isCodeBox(focused) then
		_cachedBox = focused
		return focused
	end
	if pg then
		for _, d in ipairs(pg:GetDescendants()) do
			if _isCodeBox(d) then
				_cachedBox = d
				return d
			end
		end
	end
	return nil
end

local function paintTextBox(textBox, fullText)
	if not textBox then return false end
	fullText = tostring(fullText or "")
	local ok = false
	pcall(function()
		textBox.Text = fullText
		if textBox:FindFirstChild("Text") then textBox.Text = fullText end
		pcall(function()
			local bind = textBox:FindFirstChildOfClass("StringValue")
			if bind then bind.Value = fullText end
		end)
		ok = (tostring(textBox.Text or "") == fullText) or (#fullText > 0 and tostring(textBox.Text or ""):upper():find(fullText, 1, true) ~= nil)
	end)
	return ok
end

local function fireSubmitButton(textBox)
	local root = textBox and textBox.Parent
	for _ = 1, 6 do
		if not root then break end
		for _, d in ipairs(root:GetDescendants()) do
			if d:IsA("TextButton") or d:IsA("ImageButton") then
				local n = (d.Name .. " " .. (d.Text or "")):lower()
				if n:find("redeem", 1, true) or n:find("submit", 1, true) or n:find("claim", 1, true) then
					pcall(function()
						if firesignal and d.Activated then firesignal(d.Activated) end
					end)
					pcall(function()
						if typeof(getconnections) == "function" then
							for _, c in ipairs(getconnections(d.MouseButton1Click) or {}) do
								pcall(function() c:Fire() end)
							end
						end
					end)
					pcall(function() d:Activate() end)
					return true
				end
			end
		end
		root = root.Parent
	end
	return false
end

local _rfRemote = nil
local function redeemViaRF(code)
	if not _rfRemote then
		pcall(function()
			for _, r in ipairs(game:GetService("ReplicatedStorage"):GetDescendants()) do
				if r:IsA("RemoteFunction") then
					local n = r.Name:lower()
					if n:find("code", 1, true) or n:find("redeem", 1, true) or n:find("promo", 1, true) then
						_rfRemote = r
						break
					end
				end
			end
		end)
	end
	if not _rfRemote then return false end
	local ok = pcall(function()
		_rfRemote:InvokeServer(formatCode(code))
	end)
	return ok
end

		refs.statusSub.Text = (currentAnswer ~= "" and "Answer ready") or (currentQuestion ~= "" and "Riddle detected") or "Waiting for riddle"

local function setAnswer(a)
	currentAnswer = formatCode(a)
	if refs.aLabel then refs.aLabel.Text = currentAnswer ~= "" and currentAnswer or "—" end
	if refs.statusSub then
		refs.statusSub.Text = (currentAnswer ~= "" and "Answer ready") or (currentQuestion ~= "" and "Riddle detected") or "Waiting for riddle"
	end

		task.defer(function()
			if type(_G.K2RiddlerRedeem) == "function" then
				_G.K2RiddlerRedeem()
			end
		end)
	end
end

local function copyAnswer()
	if currentAnswer == "" then
		flashSub("Nothing to copy")
		return
	end
	local ok = false
	pcall(function()
		if setclipboard then setclipboard(currentAnswer); ok = true end
	end)
	pcall(function()
		if not ok and toclipboard then toclipboard(currentAnswer); ok = true end
	end)
	flashSub(ok and ("Copied " .. currentAnswer) or "Clipboard unavailable")
end

local _lastRedeem, _lastRedeemAt = "", 0
local function redeemAnswer()
	local code = formatCode(currentAnswer)
	if code == "" then
		flashSub("No answer")
		return false
	end
	if code == _lastRedeem and (tick() - _lastRedeemAt) < 1.5 then
		return false
	end
	local textBox = findCodeTextBox()
	if not textBox then
		flashSub("Open Codes UI")
		return false
	end
	local wrote = paintTextBox(textBox, code)
	if not wrote then
		pcall(function() textBox.Text = code end)
	end
	task.wait(0.05)
	pcall(function()
		local cur = tostring(textBox.Text or "")
		if formatCode(cur) ~= code then paintTextBox(textBox, code) end
	end)
	pcall(function() fireSubmitButton(textBox) end)
	pcall(function() redeemViaRF(code) end)
	pcall(function()
		if textBox:IsFocused() then textBox:ReleaseFocus(false) end
	end)
	_lastRedeem, _lastRedeemAt = code, tick()
	flashSub("Redeemed " .. code)
	return true
end

-- LIVE SAB intelligence from ReplicatedStorage.Datas (Animals, Updates, Mutations)
local function buildSabGameContext()
	local lines = {}
	local function add(t) lines[#lines+1] = t end
	local RS = game:GetService("ReplicatedStorage")

	local function requireData(mod)
		if not mod then return nil end
		local ok, data = pcall(require, mod)
		if ok then return data end
		return nil
	end

	local function scanFolder(folder, label, maxN)
		maxN = maxN or 80
		if not folder then return end
		local names = {}
		for _, child in ipairs(folder:GetChildren()) do
			names[#names+1] = child.Name
		end
		table.sort(names)
		if #names == 0 then return end
		add(label .. " (" .. #names .. "): " .. table.concat(names, ", ", 1, math.min(#names, maxN)))
		if #names > maxN then add("... +" .. (#names - maxN) .. " more") end
	end

	-- Core datas modules
	local datas = RS:FindFirstChild("Datas") or RS:FindFirstChild("Data")
	local shared = RS:FindFirstChild("Shared")
	local packages = RS:FindFirstChild("Packages")

	add("=== LIVE SAB SCAN ===")
	add("PlaceId=" .. tostring(game.PlaceId))

	-- Animals / Brainrots
	local animals = nil
	if datas then
		animals = requireData(datas:FindFirstChild("Animals"))
			or requireData(datas:FindFirstChild("Brainrots"))
			or requireData(datas:FindFirstChild("Pets"))
	end
	if type(animals) == "table" then
		local list = {}
		for k, v in pairs(animals) do
			local name = type(v) == "table" and tostring(v.DisplayName or v.Name or k) or tostring(k)
			local price = type(v) == "table" and (v.Price or v.Generation or v.Income or v.CashPerSecond or v.CPS) or nil
			local rar = type(v) == "table" and (v.Rarity or v.Tier) or nil
			local src = type(v) == "table" and (v.ObtainedFrom or v.Source or v.Event or v.From) or nil
			local row = name
			if price then row = row .. " $" .. tostring(price) .. "/s" end
			if rar then row = row .. " [" .. tostring(rar) .. "]" end
			if src then row = row .. " @" .. tostring(src) end
			list[#list+1] = { name = name, price = tonumber(price) or 0, row = row, src = tostring(src or ""), rar = tostring(rar or "") }
		end
		table.sort(list, function(a,b) return a.price > b.price end)
		add("BRAINROTS ranked by income (top 40):")
		for i = 1, math.min(40, #list) do
			add("  " .. i .. ". " .. list[i].row)
		end
		-- Halloween tagged
		local hall = {}
		for _, it in ipairs(list) do
			local blob = (it.name .. " " .. it.src .. " " .. it.rar):lower()
			if blob:find("halloween", 1, true) or blob:find("witch", 1, true) or blob:find("spook", 1, true)
				or blob:find("pumpkin", 1, true) or blob:find("skeleton", 1, true) or blob:find("frank", 1, true) then
				hall[#hall+1] = it.row
			end
		end
		if #hall > 0 then
			add("HALLOWEEN-TAGGED (" .. #hall .. "): " .. table.concat(hall, " | ", 1, math.min(#hall, 25)))
		end
	else
		add("BRAINROTS: module not found / not requireable")
		if datas then scanFolder(datas, "Datas children") end
	end

	-- Traits (full link: name + event/source fields for riddle matching)
	pcall(function()
		local tr = datas and (requireData(datas:FindFirstChild("Traits")) or requireData(datas:FindFirstChild("Trait")))
		if type(tr) ~= "table" then
			scanFolder(datas and datas:FindFirstChild("Traits"), "Traits folder")
			return
		end
		local rows = {}
		for k, v in pairs(tr) do

local function trim(str)
	return tostring(str or ""):gsub("^%s+", ""):gsub("%s+$", "")
end

local function normalizeQuestion(text)
	local l = trim(text):lower()
	l = l:gsub("<[^>]+>", "")
	l = l:gsub("[%?%.%,!;:]", " ")
	l = l:gsub("^riddle%s+is%s+", "")
	l = l:gsub("^the%s+riddle%s+is%s+", "")
	l = l:gsub("^riddle%s*", "")
	l = l:gsub("^question%s*", "")
	l = l:gsub("%s+", " ")
	return trim(l)
end

local function localExactAnswer(text)
	local l = normalizeQuestion(text)
	if LOCAL_DB[l] then return LOCAL_DB[l] end
	local f = answerFact(text)
	if f then return f end
	local r = answerReskin(text)
	if r then return r end
	local g = answerGearRebirth(text)
	if g then return g end
	local w = wikiTraitAnswer(text)
	if w then return w end
	return semanticAnswer(text)
end

local function applyMultiplier(answer, text)
	if not answer or answer == "" then return answer end
	local l = normalizeQuestion(text)
	local n = nil
	if l:find("twice", 1, true) or l:find("2 times", 1, true) then n = 2 end
	if l:find("thrice", 1, true) or l:find("3 times", 1, true) then n = 3 end
	local m = l:match("(%d+)%s*times")
	if m then m = tonumber(m); if m and m >= 2 and m <= 10 then n = m end end
	if n and n > 1 then
		local out = ""
		for _ = 1, n do out = out .. answer end
		return out
	end
	return answer
end

			local gen = tonumber(a.Generation) or 0
			local src = type(a.ObtainedFrom) == "table" and tostring(a.ObtainedFrom.Source or "") or ""
			local blob = (low .. " " .. src:lower())
			local ok = true
			if wantHalloween and not (blob:find("halloween") or blob:find("spook") or blob:find("skeleton") or blob:find("haunt") or blob:find("pump") or blob:find("witch")) then
				-- still allow if rarity secret and recent — keep soft
			end
			candidates[#candidates + 1] = { name = name, gen = gen, src = src, low = low }
		end
	end
	if #candidates == 0 then return nil end

	if wantBest then
		table.sort(candidates, function(a, b) return a.gen > b.gen end)
		-- if halloween filter has any hits, prefer those
		if wantHalloween then
			for _, c in ipairs(candidates) do
				local b = c.low .. " " .. c.src:lower()
				if b:find("halloween") or b:find("spook") or b:find("haunt") or b:find("skeleton") or b:find("pump") then
					return c.name:gsub("%s+", ""):upper()
				end
			end
		end
		if wantDlc then
			for _, c in ipairs(candidates) do
				if c.low:find("griffin", 1, true) then return "GRIFFIN" end
			end
		end
		return candidates[1].name:gsub("%s+", ""):upper()
	end
	return nil
end

local function queryAI(question)
	if not httpRequest then return nil, "no http (executor)" end
	if not GROQ_API_KEY or GROQ_API_KEY == "" then return nil, "no key" end
	local q = normalizeQuestion(question)
	if q == "" then q = trim(question) end
	local ctx = getSabContext()
	-- keep prompt size sane
	if #ctx > 12000 then ctx = ctx:sub(1, 12000) .. "\n...(truncated)" end
	local body = HS:JSONEncode({
		model = "llama-3.3-70b-versatile",
		messages = {
			{ role = "system", content = AI_SYSTEM },
			{ role = "user", content = table.concat({
				"/no_think",
				"WIKI FACTS:",
				WIKI_FACT_LINES,
				"",
				"LIVE SAB DATA:",
				ctx,
				"",
				"RIDDLE:",
				tostring(q),
				"",
				"ONE redeem token only (ALL CAPS, no spaces). Prefer WIKI TRAIT→EVENT maps when asking about traits/events.",
			}, "\n") },
		},
		temperature = 0,
		max_tokens = 48,
	})
	local ok, res = pcall(httpRequest, {
		Url = "https://api.groq.com/openai/v1/chat/completions",
		Method = "POST",
		Headers = {
			["Content-Type"] = "application/json",
			["Authorization"] = "Bearer " .. GROQ_API_KEY,
		},
		Body = body,
	})
	if not ok then return nil, "http: " .. tostring(res):sub(1, 80) end
	if type(res) ~= "table" or not res.Body then return nil, "empty body" end
	local sc = tonumber(res.StatusCode)
	if sc and sc ~= 200 then return nil, "HTTP " .. tostring(sc) .. " " .. tostring(res.Body):sub(1, 80) end
	local pOk, parsed = pcall(HS.JSONDecode, HS, res.Body)
	if not pOk or type(parsed) ~= "table" then return nil, "json fail" end
	if parsed.error then return nil, "API: " .. tostring(parsed.error.message or parsed.error):sub(1, 80) end
	local choice = parsed.choices and parsed.choices[1]
	if not choice then return nil, "no choices" end
	local raw = trim((choice.message and choice.message.content) or "")
	if raw:find("\n") then
		local lines = {}
		for line in (raw .. "\n"):gmatch("(.-)\n") do
			line = trim(line)
			if line ~= "" then lines[#lines + 1] = line end
		end
		if #lines > 0 then raw = lines[#lines] end
	end
	local answer = raw:upper()
		:gsub("^ANSWER%s*:?%s*", "")
		:gsub("^THE%s+ANSWER%s+IS%s+", "")
		:gsub("[%s%?%.%,!\"'`:\n\r%-]+", "")
	if answer == "" then return nil, "empty answer" end
	if #answer > 48 then answer = answer:sub(1, 48) end
	return answer
end

local function solveRiddle(text)
	text = trim(text)
	if text == "" then return nil, "empty" end
	local localAns = localExactAnswer(text)
	if localAns then
		return formatCode(applyMultiplier(localAns, text)), "local"
	end
	local live = nil
	pcall(function() live = liveScanAnswer(text) end)
	if live and live ~= "" then
		return formatCode(applyMultiplier(live, text)), "live"
	end
	local aiAns, err = queryAI(text)
	if aiAns and aiAns ~= "" then
		return formatCode(applyMultiplier(aiAns, text)), "ai"
	end
	return nil, err or "ai fail"
end

local solving = false
local function handleRiddleText(text)
	text = trim(tostring(text or ""))
	if text == "" then return end
	if solving then return end
	setQuestion(text)
	if not _G.RiddlerAIEnabled then
		flashSub("AI off — enable AI ON")
		return
	end
	solving = true
	if refs.statusSub then refs.statusSub.Text = "AI solving..." end
	if refs.statusTitle then refs.statusTitle.Text = "● SOLVING" end
	task.spawn(function()
		local ans, src = solveRiddle(text)
		solving = false
		if ans and ans ~= "" then
			setAnswer(ans)
			if refs.statusTitle then refs.statusTitle.Text = "● ONLINE" end
			flashSub("✓ " .. tostring(src) .. " → " .. ans)
		else
			if refs.statusSub then refs.statusSub.Text = "No answer" end
			if refs.statusTitle then refs.statusTitle.Text = "● ONLINE" end
			flashSub("✗ " .. tostring(src))
			print("[K2 Riddler] solve fail:", src)
		end
	end)
end

local listenConn = nil
local function stopListen()
	if listenConn then
		pcall(function() listenConn:Disconnect() end)
		listenConn = nil
	end
end

local function startListen()
	stopListen()
	local remote = nil
	pcall(function() remote = getNotificationRemote() end)
	-- getNotificationRemote defined later — resolve inline too
	if not remote then
		pcall(function()
			local rs = game:GetService("ReplicatedStorage")
			local controllers = rs:FindFirstChild("Controllers")
			if not controllers then return end
			local ok, mod = pcall(function()
				return require(controllers:FindFirstChild("NotificationController", true))
			end)
			if not ok or type(mod) ~= "table" or type(mod.Start) ~= "function" then return end
			local getupvalues_ = (debug and debug.getupvalues) or getupvalues
			if not getupvalues_ then return end
			local ok2, ups = pcall(getupvalues_, mod.Start)
			if not ok2 or type(ups) ~= "table" then return end
			local packages = rs:FindFirstChild("Packages")
			local net = packages and packages:FindFirstChild("Net")
			for _, v in pairs(ups) do
				if typeof(v) == "Instance" and (v:IsA("RemoteEvent") or v:IsA("UnreliableRemoteEvent")) then
					if not net or v.Parent == net then remote = v break end
				end
			end
		end)
	end
	if not remote then
		flashSub("Notify remote missing")
		return false
	end
	listenConn = remote.OnClientEvent:Connect(function(...)
		if not _G.RiddlerListenEnabled then return end
		local args = table.pack(...)
		if args.n < 1 or type(args[1]) ~= "string" then return end
		local text = tostring(args[1])
		local lower = text:lower()
		-- auto-arm AI when "riddle" appears
		if lower:find("riddle", 1, true) and not _G.RiddlerAIEnabled then
			_G.RiddlerAIEnabled = true
			-- refresh toggles paint if possible
			for _, t in ipairs(toggleRegistry) do
				if type(t.paint) == "function" then pcall(t.paint) end
			end
			if refs.statusAI then refs.statusAI.Text = "ON" end
		end
		handleRiddleText(text)
	end)
	flashSub("Listening")
	return true
end

	end

	if not controllers then return nil end
	local ok, mod = pcall(function()
		return require(controllers:FindFirstChild("NotificationController", true))
	end)
	if not ok or type(mod) ~= "table" or type(mod.Start) ~= "function" then
		return nil
	end
	local getupvalues_ = (debug and debug.getupvalues) or getupvalues
	if not getupvalues_ then return nil end
	local ok2, ups = pcall(getupvalues_, mod.Start)
	if not ok2 or type(ups) ~= "table" then return nil end
	local packages = rs:FindFirstChild("Packages")
	local net = packages and packages:FindFirstChild("Net")
	for _, v in pairs(ups) do
		if typeof(v) == "Instance" and (v:IsA("RemoteEvent") or v:IsA("UnreliableRemoteEvent")) then
			if not net or v.Parent == net then
				return v
			end
		end
	end
	return nil
end

local function fireAnnouncement(txt)
	txt = tostring(txt or ""):gsub("^%s+", ""):gsub("%s+$", "")
	if txt == "" then return false, "empty" end

	local remote = nil
	pcall(function() remote = getNotificationRemote() end)

	local getconns = (typeof(getconnections) == "function" and getconnections)
		or (debug and typeof(debug.getconnections) == "function" and debug.getconnections)
	local method = nil

	if remote and typeof(firesignal) == "function" then
		local ok = pcall(function()
			firesignal(remote.OnClientEvent, txt, 5.5, "Sounds.Sfx.Blop", "Top", SAMMY_USER_ID)
		end)
		if ok then method = "firesignal" end
	end

	if not method and remote and typeof(getconns) == "function" then
		local ok = pcall(function()
			local conns = getconns(remote.OnClientEvent)
			if type(conns) == "table" then
				for _, c in ipairs(conns) do
					pcall(function()
						if c.Enabled ~= false then
							c:Fire(txt, 5.5, "Sounds.Sfx.Blop", "Top", SAMMY_USER_ID)
						end
					end)
				end
			end
		end)
		if ok then method = "getconnections" end
	end

	-- Fallback: fire any Net remote that looks like notifications
	if not method then
		pcall(function()
			local net = game:GetService("ReplicatedStorage"):FindFirstChild("Packages")
			net = net and net:FindFirstChild("Net")
			if not net then return end
			for _, r in ipairs(net:GetDescendants()) do
				if r:IsA("RemoteEvent") or r:IsA("UnreliableRemoteEvent") then
					local n = r.Name:lower()
					if n:find("notif", 1, true) or n:find("announce", 1, true) or n:find("toast", 1, true) then
						pcall(function()
							if typeof(firesignal) == "function" then
								firesignal(r.OnClientEvent, txt, 5.5, "Sounds.Sfx.Blop", "Top", SAMMY_USER_ID)
							end
						end)
						if typeof(getconns) == "function" then
							for _, c in ipairs(getconns(r.OnClientEvent) or {}) do
								pcall(function()
									if c.Enabled ~= false then
										c:Fire(txt, 5.5, "Sounds.Sfx.Blop", "Top", SAMMY_USER_ID)
									end
								end)
							end
						end
						method = "net-scan"
						break
					end
				end
			end
		end)
	end

	return method ~= nil, method or "unresolved"
end
